CMAKE_MINIMUM_REQUIRED(VERSION 3.16 FATAL_ERROR)
CMAKE_POLICY(VERSION 3.16)

project(QtTorch)

# Handle CMP0026 policy
cmake_policy(SET CMP0026 NEW)

# Explicitly disable Qt4
SET(QT_FOUND TRUE)
SET(QT4_FOUND FALSE)
SET(WITH_QT4 OFF CACHE BOOL "Use Qt4 - OFF to use Qt6" FORCE)

# Add local cmake modules path
SET(CMAKE_MODULE_PATH ${CMAKE_MODULE_PATH} "${CMAKE_CURRENT_SOURCE_DIR}/cmake")

# Find local qtlua
FIND_PACKAGE(QtLua REQUIRED)

FIND_PACKAGE(Torch REQUIRED)
FIND_PACKAGE(Qt6 REQUIRED COMPONENTS Core Gui Widgets)

# Include directories for qtlua
INCLUDE_DIRECTORIES(
  "${CMAKE_CURRENT_SOURCE_DIR}"
  ${QTLUA_INCLUDE_DIRS}
)
LINK_DIRECTORIES(${LUA_LIBDIR})

SET(src qttorch.h qttorch.cpp)
SET(luasrc init.lua)

ADD_TORCH_PACKAGE(qttorch "${src}" "${luasrc}")
TARGET_LINK_LIBRARIES(qttorch
  luaT
  TH
  ${QTLUA_LIBRARIES}
  Qt6::Core
  Qt6::Gui
  Qt6::Widgets
)

IF(LUALIB)
  TARGET_LINK_LIBRARIES(qttorch ${LUALIB})
ENDIF()

# Enable Qt features
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_AUTOUIC ON)


INSTALL(DIRECTORY "doc" DESTINATION "${Torch_INSTALL_LUA_PATH_SUBDIR}/qttorch")
